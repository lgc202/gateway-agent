> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 建立一个可运行、可测试的 Go 服务，通过 Fake Gateway 完成企业级 AI 网关 Agent 的第一条只读路由查询链路。

**Architecture:** 参考 OneX 采用 `apiserver + worker + controlplane + internal/pkg` 的工程结构。Slice 1A 只创建 apiserver、controlplane/gateway 和必要的 internal/pkg；apiserver 拥有 Config、Server、Gin、Wire 和启动生命周期，controlplane 保存网关业务语义，internal/pkg 保存跨组件共享的健康检查等技术能力。Slice 1A 不接模型、不接 PostgreSQL、不接真实 Higress，也不包含任何写能力。

**Tech Stack:** Go 1.24.x、Gin、Google Wire、标准库 `log/slog`、OpenAPI 3.1、Go testing/httptest、Make、GitHub Actions（或客户 CI 的等价配置）。PostgreSQL/sqlc 从 Slice 1B 引入；Buf/grpc-go 仅在出现明确 gRPC 接口时引入。

```go
package gateway

import (
	"errors"
	"strings"
)

var ErrInvalidValue = errors.New("invalid gateway domain value")

type Environment string

const (
	EnvironmentDev  Environment = "dev"
	EnvironmentTest Environment = "test"
	EnvironmentProd Environment = "prod"
)

func ParseEnvironment(value string) (Environment, error) {
	switch Environment(strings.TrimSpace(value)) {
	case EnvironmentDev:
		return EnvironmentDev, nil
	case EnvironmentTest:
		return EnvironmentTest, nil
	case EnvironmentProd:
		return EnvironmentProd, nil
	default:
		return "", ErrInvalidValue
	}
}

type GatewayID string
type Namespace string
type RouteID string

func NewGatewayID(value string) (GatewayID, error) {
	value = strings.TrimSpace(value)
	if value == "" {
		return "", ErrInvalidValue
	}
	return GatewayID(value), nil
}

func NewNamespace(value string) (Namespace, error) {
	value = strings.TrimSpace(value)
	if value == "" {
		return "", ErrInvalidValue
	}
	return Namespace(value), nil
}

func NewRouteID(value string) (RouteID, error) {
	value = strings.TrimSpace(value)
	if value == "" {
		return "", ErrInvalidValue
	}
	return RouteID(value), nil
}
```

- [ ] **Step 4: 实现最小 Fake Adapter**

```go
type Adapter struct {
	routes []gateway.Route
}

func New(routes []gateway.Route) *Adapter {
	return &Adapter{routes: append([]gateway.Route(nil), routes...)}
}

func (a *Adapter) FindRoutes(ctx context.Context, query gateway.RouteQuery) ([]gateway.Route, error) {
	if err := ctx.Err(); err != nil {
		return nil, err
	}
	result := make([]gateway.Route, 0)
	for _, route := range a.routes {
		if query.Matches(route) {
			result = append(result, route)
		}
	}
	return result, nil
}
```

将 `Matches` 规则放在领域查询对象中，避免不同 Adapter 对同一查询语义各自解释。

```go
type Status string

const (
	StatusResolved  Status = "resolved"
	StatusAmbiguous Status = "ambiguous"
)

var ErrRouteNotFound = errors.New("route not found")

type Result struct {
	Status     Status
	Route      *gateway.Route
	Candidates []gateway.Route
}

type Service struct {
	reader gateway.Reader
}

func New(reader gateway.Reader) *Service {
	return &Service{reader: reader}
}

func (s *Service) Execute(ctx context.Context, query gateway.RouteQuery) (Result, error) {
	routes, err := s.reader.FindRoutes(ctx, query)
	if err != nil {
		return Result{}, fmt.Errorf("find routes: %w", err)
	}
	switch len(routes) {
	case 0:
		return Result{}, ErrRouteNotFound
	case 1:
		route := routes[0]
		return Result{Status: StatusResolved, Route: &route}, nil
	default:
		return Result{Status: StatusAmbiguous, Candidates: routes}, nil
	}
}
```
